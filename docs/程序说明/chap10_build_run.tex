\section{构建、运行与并行}
\label{sec:build}

\subsection{编译}
\begin{lstlisting}
make                 # 在仓库根目录；产物 bin/MultiPhysMixSolver
                     # （同时建兼容软链 bin/opencfd-ec1.16a.out）
make -j4             # 并行编译（模块依赖见 src/modules.dep，首次构建同样安全）
make clean           # 删除 build/（*.o *.mod）与 bin/
make info            # 打印编译器 / 选项 / 目标
\end{lstlisting}
源码与中间产物分离：\code{src/} 只放源码 \code{*.f90}（另有兼容入口 \code{src/makefile}）；
\code{build/} 存放 \code{*.o}、\code{*.mod}（由 make 自动创建，可随时删除）；
\code{bin/} 存放可执行文件。旧的 \code{cd src \&\& make} 仍然可用（转发到顶层 \code{makefile}）。
当前编译选项（顶层 \code{makefile}）：
\begin{lstlisting}
f77 = mpif90
opt = -g -std=legacy -ffree-line-length-none -freal-4-real-8 -O3
\end{lstlisting}
其中 \code{-freal-4-real-8} 把单精度提升为双精度（\code{real(PRE\_EC)}），
\code{-std=legacy} 兼容旧代码风格。需要 OpenMP 时用 \code{-openmp} 并把 \code{NUM\_THREADS} 设为线程数。
\textbf{改动任何 \code{src/*.f90} 后必须重新 \code{make} 再验证。}
若新增源文件或改动了 \code{use}/\code{module} 关系，先运行
\code{bash util/gen\_moddep.sh} 重新生成 \code{src/modules.dep}
（顶层 makefile 据此确定编译顺序，\code{make -j} 才可靠）。

\subsection{运行}
\begin{lstlisting}
cd cases/<case>
mpirun -np 1 ../../bin/MultiPhysMixSolver   # 交错耦合（码 12/19）必须单进程
mpirun -np 4 ../../bin/MultiPhysMixSolver   # 多进程：需块数 >= 进程数，自动分区
\end{lstlisting}
\begin{itemize}
  \item 可执行文件统一在 \code{bin/}，算例按相对路径调用 \code{../../bin/MultiPhysMixSolver}；
        \code{bin/} 下同时有兼容软链 \code{opencfd-ec1.16a.out -> MultiPhysMixSolver}，
        旧作业脚本若写 \code{./opencfd-ec1.16a.out}，把该软链拷到算例目录即可；
  \item 启动会依次做：读 \code{control.ec} $\to$ \code{prepare\_bc3d\_input}（校验/生成 \code{bc3d.inp}）
        $\to$ 网格质量检查（含\textbf{单元体积符号}：任一块出现负体积 = 左手索引三元组，
        直接打印块号/最小体积位置并 \code{MPI\_Abort}，见第 \ref{sec:ac-mblock} 节）
        $\to$ \code{Init} $\to$ 读/生成 \code{wall\_dist.dat}
        $\to$ \code{Init\_flow}（有 \code{field\_restart.dat} 则续算）$\to$ 时间推进；
  \item 多进程时会自动生成 \code{partation-auto.dat}、\code{part\_grid.dat}；
        若目录里已有 \code{partation.dat} 则优先使用；
  \item \textbf{交错耦合模式目录里不要放 \code{partation.dat}}（否则可能被分到多进程而挂死）。
\end{itemize}

\paragraph{编译/运行环境注意事项（集群经验）}
\begin{itemize}
  \item \textbf{编译器的严格程度会暴露代码里违反标准之处}：例如“同一文件同时连到两个 unit”
        在新版 libgfortran 下被放宽（不报错），但旧版会致命报错
        \code{File already opened in another unit}。本仓库已在 2026-09-25 修复
        \code{control.ec} 的该问题；换工具链后若报类似信息，优先按此思路排查
        （\code{open} 同一文件两次/两个 unit）。
  \item \textbf{集群上通常是另一份源码副本}：改完要\textbf{同步 \code{src/}（必要时连同
        \code{makefile}、\code{src/modules.dep}）并重新 \code{make}}，
        再把 \code{bin/MultiPhysMixSolver}（或兼容软链 \code{opencfd-ec1.16a.out}）拷到算例目录；
        不要只拷二进制而忘了源码版本一致。
  \item \textbf{保留 \code{field\_restart.dat}}：作业被时限杀死后可从最近保存点续算；
        建议把 \code{Kstep\_restart} 设小一些（如 100$\sim$500）以限制重算量。
  \item 作业脚本示例（PBS/Slurm 通用骨架）：
\begin{lstlisting}
cd $CASE_DIR
module load mpi/...          # 与编译时一致的 MPI
mpirun -np 1 ../../bin/MultiPhysMixSolver > run.log 2>&1
\end{lstlisting}
\end{itemize}

\subsection{集群（Slurm）作业与本地/集群同步}
集群入口 \code{ssh wuzh02}，源码副本固定在
\code{/work/home/lijunyang/sundong/MultiPhysMixSolver}。本地与集群用
\code{util/sync\_cluster.sh} 保持一致：
\begin{lstlisting}
bash util/sync_cluster.sh push   # 本地 -> 集群：代码/输入/文档/记忆库（排除 .git、build/、bin/、
                                 #  *.vtk、flow3d*.dat、field_restart.dat、Mesh3d.dat、np*/ 等）
bash util/sync_cluster.sh pull   # 集群 -> 本地：只取 memory-bank/、docs/、日志、Residual.dat、
                                 # control.ec、output_para.out 与分析输出（VTK/重启留在集群）
\end{lstlisting}
\textbf{作业一律 \code{sbatch} 提到计算节点}（不占登录节点）；测试优先在集群做，
后处理（\code{analyze\_*.py}、画图）可同步回本地；\textbf{每次集群运行的现场控制参数
（\code{np*/control.ec}、\code{output\_para.out}）必须 \code{pull} 回本地留存}。
\textbf{改完 \code{src/*.f90} 必须在集群重新 \code{make -j4} 通过再提交}：
集群工具链是 gfortran 7.3.1 $+$ hpcx 2.11，比本地新版编译器严格，容易出现本地不报、
集群致命的标准违规（见上一节“双 unit”经验）。

以 \code{cases/lidcavity\_ac\_3blk} 为例（Slurm 分区 \code{wzacnormal}）：
\begin{lstlisting}
cd cases/lidcavity_ac_3blk
./prep.sh 10000                  # 生成 Mesh3d.x/bc3d.inp/partation_np{2,3}.dat -> np1 np2 np3
                                 # 参数 = 总步数（t_end=Kstep_save=步数、Kstep_show=步数/5）；
                                 # 不给参数则各目录软链算例自带 control.ec（1000 步冒烟）
cd np1 && sbatch -n 1 ../run.slurm
cd ../np2 && sbatch -n 2 ../run.slurm     # partation_np2 = {1,2}|{3}
cd ../np3 && sbatch -n 3 ../run.slurm     # partation_np3 = 1/1/1
\end{lstlisting}
\begin{itemize}
  \item \code{sbatch -n} 必须等于该运行目录 \code{partation.dat} 的进程数；\code{run.slurm}
        用 \code{\$SLURM\_NTASKS} 调 \code{mpirun}，日志为 \code{run\_out.log}/\code{run\_err.log}；
  \item 运行目录比算例目录多一层，二进制路径是 \code{../../../bin/MultiPhysMixSolver}
        （\code{run.slurm} 内部 \code{cd ../../..} 定位并检查可执行位，路径写错会立刻
        \code{mpirun ... could not access or execute an executable}）；
  \item 登录节点 \code{python3}（3.6.8）没有 numpy：先
        \code{module load apps/anaconda3/2023.09}（Python 3.11.5 $+$ numpy 1.24.3）再跑
        \code{analyze\_*.py}；或把 \code{flow3d\_block\_*.vtk}（约 190 kB/块）拉回本地分析；
  \item 作业被杀后目录里仍保留 \code{field\_restart.dat}，直接重提交即可续算。
\end{itemize}

\paragraph{实测（2026-10-18，wuzh02）}同一份源码在集群 clean 重编（38 个目标、0 error、
2 warning、68 s）后，三块方腔 $10^{4}$ 步：墙钟 np$=$1/2/3 为 27/18/8 s；
np$=$2、np$=$3 对 np$=$1 的差为 RMS$(u)=5.77\times10^{-7}$、$\max|\Delta u|=1.16\times10^{-6}$，
且集群 np$=$1 的流场与本地 \code{q1\_10k} \textbf{逐位相同}（RMS$=0$），
说明结果与编译器版本、进程数无关。

\subsection{调试手段}
\begin{itemize}
  \item \code{IF\_Debug=1}：打印接口配对诊断（\code{dbg\_dump\_interfaces}）；
  \item \code{Pdebug(1:4)}：分块/残差的额外打印级别；
  \item \code{Iflag\_bc\_check=2}：边界文件严格校验，便于定位 \code{bc3d.inp} 与
        \code{bc3d\_interface.inp} 不一致；
  \item \code{Kstep\_show}、\code{AC\_Print}：控制残差输出频率；
  \item 多进程时优先看 0 号进程日志；挂死常见原因是“各 rank 的开关取值不一致”
        （新开关必须进 \code{bcast\_para}）。
\end{itemize}
